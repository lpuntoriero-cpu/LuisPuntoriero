const UART_SERVICE_UUID = "6e400001-b5a3-f393-e0a9-e50e24dcca9e"; const UART_TX_CHARACTERISTIC = "6e400002-b5a3-f393-e0a9-e50e24dcca9e"; const UART_RX_CHARACTERISTIC = "6e400003-b5a3-f393-e0a9-e50e24dcca9e"; let gattServer = null; let txCharacteristic = null; function log(msg) { const c = document.getElementById("console"); c.innerHTML += "
> " + msg; c.scrollTop = c.scrollHeight; } async function conectarBLE() { try { log("Buscando Micro:bit..."); const device = await navigator.bluetooth.requestDevice({ filters: [{ namePrefix: 'BBC micro:bit' }], optionalServices: [UART_SERVICE_UUID] }); log("Conectando..."); gattServer = await device.gatt.connect(); const service = await gattServer.getPrimaryService(UART_SERVICE_UUID); txCharacteristic = await service.getCharacteristic(UART_TX_CHARACTERISTIC); // Intentar habilitar notificaciones RX si la placa las soporta try { const rxCharacteristic = await service.getCharacteristic(UART_RX_CHARACTERISTIC); await rxCharacteristic.startNotifications(); rxCharacteristic.addEventListener('characteristicvaluechanged', (event) => { const decoder = new TextDecoder(); log("Microbit: " + decoder.decode(event.target.value)); }); } catch (e) { // Ignorar si RX no está activo para evitar bloqueos } document.getElementById("lblEstado").innerText = "Conectado"; document.getElementById("lblEstado").classList.add("connected"); document.getElementById("btnLargada2").disabled = false; document.getElementById("btnLargada1").disabled = false; document.getElementById("btnReset").disabled = false; log("¡Conexión establecida!"); device.addEventListener('gattserverdisconnected', onDisconnected); } catch (error) { log("Error: " + error.message); } } let enviandoComando = false; async function enviarComando(cmd) { if (!txCharacteristic) { log("Error: Sin conexión"); return; } // Evitar que se encabriten múltiples clics seguidos if (enviandoComando) return; enviandoComando = true; try { const encoder = new TextEncoder(); const data = encoder.encode(cmd + "\n"); // Intentar primero con writeValueWithoutResponse (es más fluido en BLE) try { await txCharacteristic.writeValueWithoutResponse(data); } catch (err) { // Si falla, usar el método con respuesta segura await txCharacteristic.writeValueWithResponse(data); } log("Enviado: " + cmd); } catch (error) { log("Falla GATT: " + error.message); } finally { // Dar un pequeño respiro al buffer de la placa setTimeout(() => { enviandoComando = false; }, 150); } } function onDisconnected() { document.getElementById("lblEstado").innerText = "Desconectado"; document.getElementById("lblEstado").classList.remove("connected"); document.getElementById("btnLargada2").disabled = true; document.getElementById("btnLargada1").disabled = true; document.getElementById("btnReset").disabled = true; log("Desconectado de la placa"); }
